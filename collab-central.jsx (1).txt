// ─── STRESS DETECTION ───────────────────────────────────────────────────────
const STRESS_KEYWORDS = ["stressed","anxious","burnout","panic","overwhelmed","scared","worried","depressed","hopeless","exhausted","dying","crying","failing","lost"];
function detectStress(text) {
  const lower = text.toLowerCase();
  return STRESS_KEYWORDS.some(k => lower.includes(k));
}

// ─── UTILS ──────────────────────────────────────────────────────────────────
function timeAgo(ts) {
  const diff = Date.now() - ts;
  if (diff < 60000) return "just now";
  if (diff < 3600000) return `${Math.floor(diff/60000)}m ago`;
  if (diff < 86400000) return `${Math.floor(diff/3600000)}h ago`;
  return `${Math.floor(diff/86400000)}d ago`;
}
function calcEmployability(user) {
  return Math.min(100, Math.round((user.points/5) + (user.rating*8) + ((user.helped||0)*1.5)));
}
function categoryColor(cat) {
  const map = { "Academic Doubt":"#6C63FF", "Stress Help":"#FF6584", "Project Help":"#43C6AC", "Career Confusion":"#F7971E", "Urgent Help":"#FF4757" };
  return map[cat] || "#6C63FF";
}
function stressEmoji(lvl) {
  if (lvl <= 1) return "😊";
  if (lvl <= 2) return "🙂";
  if (lvl <= 3) return "😐";
  if (lvl <= 4) return "😰";
  return "😱";
}

// ─── MAIN APP ────────────────────────────────────────────────────────────────
export default function CollabCentral() {
  const [currentUser, setCurrentUser] = useState(null);
  const [page, setPage] = useState("login");
  const [users, setUsers] = useState(SEED_USERS);
  const [requests, setRequests] = useState(SEED_REQUESTS);
  const [sessions, setSessions] = useState(SEED_SESSIONS);
  const [bookings, setBookings] = useState(SEED_BOOKINGS);
  const [chats, setChats] = useState(SEED_CHATS);
  const [reviews, setReviews] = useState(SEED_REVIEWS);
  const [activeChatKey, setActiveChatKey] = useState(null);
  const [notification, setNotification] = useState(null);
  const [modal, setModal] = useState(null);

  const notify = (msg, type="success") => {
    setNotification({msg,type});
    setTimeout(()=>setNotification(null), 3500);
  };

  const login = (user) => {
    setCurrentUser(user);
    setPage(user.role === "junior" ? "junior-dashboard" : "senior-dashboard");
  };

  const logout = () => { setCurrentUser(null); setPage("login"); };

  const sendMessage = (key, text) => {
    const newMsg = { id:"m"+Date.now(), from:currentUser.id, text, time:Date.now() };
    setChats(prev => ({ ...prev, [key]: [...(prev[key]||[]), newMsg] }));
  };

  const acceptRequest = (reqId) => {
    setRequests(prev => prev.map(r => r.id===reqId ? {...r, status:"accepted", seniorId:currentUser.id} : r));
    notify("Request accepted! Chat started.");
  };

  const resolveRequest = (reqId) => {
    setRequests(prev => prev.map(r => r.id===reqId ? {...r, status:"resolved"} : r));
    setUsers(prev => prev.map(u => u.id===currentUser.id ? {...u, points:u.points+10, helped:(u.helped||0)+1} : u));
    notify("+10 points earned! 🎉");
  };

  const addRequest = (data) => {
    const req = { id:"r"+Date.now(), ...data, juniorId:currentUser.id, seniorId:null, status:"open", createdAt:Date.now() };
    setRequests(prev => [req, ...prev]);
    return req;
  };

  const addSession = (data) => {
    const s = { id:"s"+Date.now(), ...data, seniorId:currentUser.id, participants:[] };
    setSessions(prev => [s, ...prev]);
    notify("Group session created! 🚀");
  };

  const joinSession = (sessId) => {
    setSessions(prev => prev.map(s => s.id===sessId && !s.participants.includes(currentUser.id) ? {...s, participants:[...s.participants, currentUser.id]} : s));
    notify("Joined session! See you there 🎓");
  };

  const addBooking = (data) => {
    const b = { id:"b"+Date.now(), ...data, juniorId:currentUser.id, status:"upcoming" };
    setBookings(prev => [b, ...prev]);
    notify("Session booked! 📅");
  };

  const addReview = (data) => {
    const rv = { id:"rv"+Date.now(), ...data, juniorId:currentUser.id };
    setReviews(prev => [rv, ...prev]);
    const bonus = data.stars >= 4 ? data.stars * 5 : 0;
    setUsers(prev => prev.map(u => u.id===data.seniorId ? {
      ...u,
      points: u.points + bonus,
      rating: parseFloat((((u.rating * (u.helped||1)) + data.stars) / ((u.helped||1)+1)).toFixed(1))
    } : u));
    notify(`Review submitted! Senior earned +${bonus} points 🌟`);
  };

  const registerUser = (data) => {
    const u = { id:"u"+Date.now(), ...data, points:0, rating:0, avatar:data.name.split(" ").map(n=>n[0]).join("").slice(0,2).toUpperCase(), badge: data.role==="senior"?"Rising Mentor":"New Explorer", helped:0 };
    setUsers(prev => [...prev, u]);
    return u;
  };

  const getChatKey = (id1, id2) => [id1,id2].sort().join("-");
  const getSenior = (id) => users.find(u=>u.id===id);
  const getJunior = (id) => users.find(u=>u.id===id);
  const seniors = users.filter(u=>u.role==="senior");

  const props = { currentUser, users, setUsers, requests, sessions, bookings, chats, reviews, page, setPage, login, logout, notify, sendMessage, acceptRequest, resolveRequest, addRequest, addSession, joinSession, addBooking, addReview, registerUser, getChatKey, getSenior, getJunior, seniors, activeChatKey, setActiveChatKey, modal, setModal };

  return (
    <>
      <style>{css}</style>
      {notification && <Notification {...notification} />}
      {modal && <Modal modal={modal} setModal={setModal} props={props} />}
      {!currentUser ? <LoginPage {...props} /> : (
        <AppShell {...props}>
          {page==="junior-dashboard" && <JuniorDashboard {...props} />}
          {page==="senior-dashboard" && <SeniorDashboard {...props} />}
          {page==="chat" && <ChatPage {...props} />}
          {page==="book-session" && <BookSession {...props} />}
          {page==="group-sessions" && <GroupSessions {...props} />}
          {page==="leaderboard" && <Leaderboard {...props} />}
          {page==="profile" && <Profile {...props} />}
        </AppShell>
      )}
    </>
  );
}

// ─── NOTIFICATION ────────────────────────────────────────────────────────────
function Notification({ msg, type }) {
  const bg = type==="success" ? G.success : type==="error" ? G.danger : G.accent;
  return (
    <div style={{ position:"fixed", top:20, right:20, zIndex:9999, background:bg, color:"white", padding:"12px 18px", borderRadius:10, fontSize:14, fontWeight:500, boxShadow:"0 8px 30px rgba(0,0,0,0.4)", animation:"bounceIn 0.3s ease", maxWidth:320 }}>
      {msg}
    </div>
  );
}

// ─── MODAL ───────────────────────────────────────────────────────────────────
function Modal({ modal, setModal, props }) {
  const close = () => setModal(null);
  return (
    <div className="modal-overlay" onClick={e=>e.target===e.currentTarget&&close()}>
      <div className="modal">
        {modal.type==="rate" && <RateModal {...props} modal={modal} close={close} />}
        {modal.type==="new-request" && <NewRequestModal {...props} close={close} />}
        {modal.type==="new-session" && <NewSessionModal {...props} close={close} />}
        {modal.type==="view-request" && <ViewRequestModal {...props} modal={modal} close={close} />}
      </div>
    </div>
  );
}

// ─── CHAT PAGE ────────────────────────────────────────────────────────────────
function ChatPage({ currentUser, users, chats, requests, sendMessage, setActiveChatKey, activeChatKey, getChatKey }) {
  const [text, setText] = useState("");
  const bottomRef = useRef(null);
  const otherUsers = users.filter(u=>u.id!==currentUser.id);

  // Build chat list from existing chats
  const chatKeys = Object.keys(chats).filter(k=>k.includes(currentUser.id));
  const activeKey = activeChatKey || chatKeys[0] || null;

  const getOtherUserId = (key) => key.split("-").find(id=>id!==currentUser.id);
  const getOtherUser = (key) => users.find(u=>u.id===getOtherUserId(key));

  const messages = activeKey ? (chats[activeKey]||[]) : [];
  const otherUser = activeKey ? getOtherUser(activeKey) : null;

  useEffect(()=>{ bottomRef.current?.scrollIntoView({behavior:"smooth"}); }, [messages]);

  const handleSend = () => {
    if (!text.trim() || !activeKey) return;
    sendMessage(activeKey, text.trim());
    setText("");
  };

  const quickReplies = currentUser.role==="senior"
    ? ["📝 I'll share notes!", "💪 You've got this!", "Let me explain with an example:", "Try this approach:"]
    : ["Thanks for explaining!", "Can you clarify this part?", "That makes sense now 🙌", "Could you share a resource?"];

  return (
    <div style={{ display:"flex", height:"calc(100vh - 80px)", gap:16 }} className="fade-in">
      {/* Sidebar: conversations */}
      <div className="card" style={{ width:240, padding:0, overflow:"hidden", display:"flex", flexDirection:"column" }}>
        <div style={{ padding:"16px 14px 12px", borderBottom:`1px solid ${G.border}` }}>
          <div style={{ fontWeight:600, fontSize:15 }}>Messages</div>
        </div>
        <div style={{ flex:1, overflowY:"auto" }}>
          {chatKeys.length === 0 && (
            <div style={{ padding:20, textAlign:"center", color:G.textMuted, fontSize:13 }}>No chats yet</div>
          )}
          {chatKeys.map(key=>{
            const other = getOtherUser(key);
            const msgs = chats[key]||[];
            const last = msgs[msgs.length-1];
            return (
              <div key={key} onClick={()=>setActiveChatKey(key)} style={{ padding:"12px 14px", cursor:"pointer", background:activeKey===key?`${G.accent}15`:"transparent", borderLeft:activeKey===key?`2px solid ${G.accent}`:"2px solid transparent", display:"flex", gap:10, alignItems:"center" }}>
                <div className="avatar avatar-sm" style={{ background:`linear-gradient(135deg,${G.accent},${G.pink})`, color:"white", flexShrink:0 }}>{other?.avatar||"?"}</div>
                <div style={{ flex:1, overflow:"hidden" }}>
                  <div style={{ fontSize:13, fontWeight:600, overflow:"hidden", textOverflow:"ellipsis", whiteSpace:"nowrap" }}>{other?.name?.split(" ")[0]}</div>
                  <div style={{ fontSize:11, color:G.textMuted, overflow:"hidden", textOverflow:"ellipsis", whiteSpace:"nowrap" }}>{last?.text?.slice(0,30)}...</div>
                </div>
              </div>
            );
          })}
        </div>
      </div>

      {/* Chat area */}
      <div className="card" style={{ flex:1, padding:0, display:"flex", flexDirection:"column", overflow:"hidden" }}>
        {!activeKey ? (
          <div style={{ flex:1, display:"flex", alignItems:"center", justifyContent:"center", color:G.textMuted }}>Select a conversation</div>
        ) : (
          <>
            {/* Chat header */}
            <div style={{ padding:"14px 18px", borderBottom:`1px solid ${G.border}`, display:"flex", alignItems:"center", gap:12 }}>
              <div className="avatar avatar-md" style={{ background:`linear-gradient(135deg,${G.accent},${G.pink})`, color:"white" }}>{otherUser?.avatar}</div>
              <div>
                <div style={{ fontWeight:600 }}>{otherUser?.name}</div>
                <div style={{ fontSize:12, color:G.teal }}>● Online · {otherUser?.dept}, {otherUser?.year} Year</div>
              </div>
            </div>

            {/* Messages */}
            <div style={{ flex:1, overflowY:"auto", padding:16, display:"flex", flexDirection:"column", gap:10 }}>
              {messages.map(msg=>{
                const isMine = msg.from === currentUser.id;
                return (
                  <div key={msg.id} style={{ display:"flex", flexDirection:"column", alignItems:isMine?"flex-end":"flex-start" }}>
                    <div className={`chat-bubble ${isMine?"mine":"theirs"}`}>{msg.text}</div>
                    <div style={{ fontSize:11, color:G.textMuted, marginTop:3, paddingInline:4 }}>{timeAgo(msg.time)}</div>
                  </div>
                );
              })}
              <div ref={bottomRef} />
            </div>

            {/* Quick replies */}
            <div style={{ padding:"8px 16px", display:"flex", gap:6, flexWrap:"wrap", borderTop:`1px solid ${G.border}` }}>
              {quickReplies.map(r=>(
                <button key={r} className="btn btn-ghost btn-sm" onClick={()=>setText(prev=>prev+r+" ")} style={{ fontSize:11 }}>{r}</button>
              ))}
            </div>

            {/* Input */}
            <div style={{ padding:"12px 16px", borderTop:`1px solid ${G.border}`, display:"flex", gap:10 }}>
              <input className="input" placeholder="Type a message..." value={text} onChange={e=>setText(e.target.value)} onKeyDown={e=>e.key==="Enter"&&handleSend()} style={{ flex:1 }} />
              <button className="btn btn-primary" onClick={handleSend} style={{ flexShrink:0 }}>Send →</button>
            </div>
          </>
        )}
      </div>
    </div>
  );
}

                  </div>

                  {/* Participant avatars */}
